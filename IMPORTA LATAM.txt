def nome_csv_saida_para(pdf_path: str) -> str:
    data_proc = datetime.now().strftime("%d%m%Y_%H%M%S")
    dir_pdf = os.path.dirname(pdf_path)
    base_pdf = os.path.splitext(os.path.basename(pdf_path))[0]

    dir_env = os.environ.get("AERO_OUTPUT_DIR", "").strip().strip('"')
    if dir_env:
        os.makedirs(dir_env, exist_ok=True)
        return os.path.join(dir_env, f"{base_pdf}_PRIMEIRA_VERSAO_{data_proc}.csv")

    dir_out = dir_pdf.replace("Escalas_Executadas", "Auditoria_Calculos")
    os.makedirs(dir_out, exist_ok=True)
    return os.path.join(dir_out, f"{base_pdf}_PRIMEIRA_VERSAO_{data_proc}.csv")

def _serie_coluna_por_posicao(df: pd.DataFrame, pos: int) -> pd.Series:
    """Retorna a série da coluna pela posição, ou vazia se não existir."""
    if pos < 0 or pos >= len(df.columns):
        return pd.Series([""] * len(df), index=df.index, dtype="object")
    col = df.columns[pos]
    return df[col].fillna("").astype(str)


def _normalizar_data_base(v: str) -> str:
    """Converte o valor da coluna 0 para dd/mm/aaaa (sem hora)."""
    if v is None:
        return ""
    s = str(v).strip()
    if s in {"", "-", "nan", "NaN"}:
        return ""

    s_up = s.upper()
    repl = {
        "DEZ": "DEC", "FEV": "FEB", "ABR": "APR", "MAI": "MAY",
        "AGO": "AUG", "SET": "SEP", "OUT": "OCT",
    }
    for k, x in repl.items():
        s_up = s_up.replace(k, x)

    formatos = [
        "%d/%m/%Y", "%d-%m-%Y", "%d/%m/%y", "%d-%m-%y",
        "%d%b%Y", "%d%b%y", "%Y-%m-%d", "%Y/%m/%d",
        "%d-%b-%Y", "%d-%b-%y",
    ]

    for fmt in formatos:
        try:
            dt = datetime.strptime(s_up, fmt)
            return dt.strftime("%d/%m/%Y")
        except Exception:
            pass

    try:
        dt = pd.to_datetime(s_up, errors="coerce", dayfirst=True)
        if pd.notna(dt):
            return dt.strftime("%d/%m/%Y")
    except Exception:
        pass

    return ""


def _compor_data_hora(data_base: str, hhmm: str) -> str:
    data_base = (data_base or "").strip()
    hhmm = (hhmm or "").strip()
    if not data_base or not hhmm:
        return ""
    return f"{data_base} {hhmm}"


def _primeiros_n(v: str, n: int) -> str:
    s = str(v or "").strip()
    return s[:n] if s else ""


def _ultimos_n(v: str, n: int) -> str:
    s = str(v or "").strip()
    return s[-n:] if s else ""


def _parse_hora_e_deslocamento(v: str):
    """Extrai hora e deslocamento de dia de valores como '04:20(+1)'.

    Retorna (hora, dias, bruto_limpo).
    Se não houver deslocamento, dias = 0.
    """
    s = str(v or "").strip()
    if not s:
        return "", 0, ""

    s = s.replace(" ", "")
    m = re.search(r"(?P<hhmm>\d{1,2}:\d{2})(?:\(\+?(?P<dias>\d+)\))?", s)
    if not m:
        return s, 0, s

    hhmm = m.group("hhmm")
    dias = int(m.group("dias") or 0)
    if len(hhmm) == 4:
        hhmm = "0" + hhmm
    return hhmm, dias, s


def _compor_data_hora_deslocada(data_base: str, hora: str, dias: int = 0) -> str:
    """Compoe data + hora aplicando deslocamento de dias."""
    data_base = (data_base or "").strip()
    hora = (hora or "").strip()
    if not data_base or not hora:
        return ""

    try:
        dt_base = datetime.strptime(data_base, "%d/%m/%Y")
    except Exception:
        return f"{data_base} {hora}" if hora else ""

    try:
        dt_final = dt_base + pd.Timedelta(days=int(dias or 0))
        return f"{dt_final.strftime('%d/%m/%Y')} {hora}"
    except Exception:
        return f"{data_base} {hora}"


def _montar_layout_latam_por_posicao(df: pd.DataFrame, return_auditoria: bool = False):
    """
    Regras solicitadas:
    - Activity = col 4; se em branco, usar col 2
    - Checkin  = data(col 0 convertida) + ' ' + 5 primeiros chars da col 3
    - Start    = data(col 0 convertida) + ' ' + 5 últimos chars da col 7
    - Dep      = 3 primeiros chars da col 7
    - Arr      = 3 primeiros chars da col 8
    - End      = data(col 0 convertida) + ' ' + 5 últimos chars da col 8
    - Checkout = data(col 0 convertida) + ' ' + 5 primeiros chars da col 9
    - AcVer    = col 13
    - DD       = em branco
    - CAT      = col 6
    - Crew     = em branco
    """
    c0 = _serie_coluna_por_posicao(df, 0)
    c2 = _serie_coluna_por_posicao(df, 2)
    c3 = _serie_coluna_por_posicao(df, 3)
    c4 = _serie_coluna_por_posicao(df, 4)
    c6 = _serie_coluna_por_posicao(df, 6)
    c7 = _serie_coluna_por_posicao(df, 7)
    c8 = _serie_coluna_por_posicao(df, 8)
    c9 = _serie_coluna_por_posicao(df, 9)
    c13 = _serie_coluna_por_posicao(df, 13)

    # Propaga a data-base para linhas de continuação do mesmo dia.
    c0_base = c0.replace(r"^\s*$", pd.NA, regex=True)
    c0_base = c0_base.replace({"-": pd.NA, "nan": pd.NA, "NaN": pd.NA})
    c0_base = c0_base.ffill().fillna("")
    data_conv = c0_base.apply(_normalizar_data_base)

    c3_hora_dia = c3.apply(_parse_hora_e_deslocamento)
    c7_hora_dia = c7.apply(_parse_hora_e_deslocamento)
    c8_hora_dia = c8.apply(_parse_hora_e_deslocamento)
    c9_hora_dia = c9.apply(_parse_hora_e_deslocamento)

    c3_hora = c3_hora_dia.apply(lambda t: t[0])
    c3_dias = c3_hora_dia.apply(lambda t: t[1])
    c7_hora = c7_hora_dia.apply(lambda t: t[0])
    c7_dias = c7_hora_dia.apply(lambda t: t[1])
    c8_hora = c8_hora_dia.apply(lambda t: t[0])
    c8_dias = c8_hora_dia.apply(lambda t: t[1])
    c9_hora = c9_hora_dia.apply(lambda t: t[0])
    c9_dias = c9_hora_dia.apply(lambda t: t[1])

    # Quando a coluna 3 carrega (+1), (+2), etc., ela define o dia-base da linha.
    # Start deve seguir apenas essa referência da linha.
    # End/Checkout podem usar o deslocamento próprio da célula, se existir.
    base_dia_linha = pd.to_numeric(c3_dias, errors="coerce").fillna(0).astype(int)
    has_new_date = c0_base.astype(str).str.strip().ne("")

    # Propaga deslocamento de forma condicional (sem usar comparações com NA)
    dias_array = base_dia_linha.values.copy()
    for idx in range(1, len(dias_array)):
        # Se linha atual tem data nova e offset=0, mantém 0 (não herda)
        if has_new_date.iloc[idx] and dias_array[idx] == 0:
            dias_array[idx] = 0
        # Se linha é continuação (sem data) e tem offset anterior, herda
        elif not has_new_date.iloc[idx] and dias_array[idx] == 0:
            dias_array[idx] = dias_array[idx - 1]

    base_dia_linha = pd.Series(dias_array, index=base_dia_linha.index).astype(int)

    start_dias = base_dia_linha
    end_dias = c8_dias.where(c8_dias > 0, base_dia_linha)
    checkout_dias = c9_dias.where(c9_dias > 0, base_dia_linha)

    out = pd.DataFrame(index=df.index)
    out["Activity"] = c4.where(c4.astype(str).str.strip() != "", c2)

    starts = []
    ends = []
    checkins = []
    checkouts = []
    prev_end_dt = None

    for idx, (d, ch, ch_days, st, st_days, ed, ed_days, co, co_days) in enumerate(zip(
        data_conv, c3_hora, c3_dias, c7_hora, start_dias, c8_hora, end_dias, c9_hora, checkout_dias
    )):
        checkin_val = ""
        if has_new_date.iloc[idx] and str(ch).strip():
            checkin_val = _compor_data_hora_deslocada(d, ch, ch_days)
        checkins.append(checkin_val)

        start_val = _compor_data_hora_deslocada(d, st, st_days)
        end_val = _compor_data_hora_deslocada(d, ed, ed_days)
        checkout_val = _compor_data_hora_deslocada(d, co, co_days)

        # Se o Checkin está em branco, estamos na mesma jornada da linha anterior.
        # Nesse caso, o Start precisa ficar depois do End anterior, podendo avançar a data.
        if not checkin_val and prev_end_dt is not None:
            start_dt = pd.to_datetime(start_val, errors="coerce", dayfirst=True)
            if pd.notna(start_dt):
                while start_dt <= prev_end_dt:
                    start_dt += pd.Timedelta(days=1)
                start_val = start_dt.strftime("%d/%m/%Y %H:%M")
                # Para continuidade de jornada, End e Checkout devem seguir a mesma data-base do Start
                # sem aplicar os deslocamentos (+1)/(+2) das colunas 8 e 9.
                base_date_str = start_dt.strftime("%d/%m/%Y")
                end_val = _compor_data_hora_deslocada(base_date_str, ed, 0)
                checkout_val = _compor_data_hora_deslocada(base_date_str, co, 0)

        starts.append(start_val)
        ends.append(end_val)
        checkouts.append(checkout_val)

        prev_end_dt = pd.to_datetime(end_val, errors="coerce", dayfirst=True)

    out["Start"] = starts
    out["End"] = ends

    out["Checkin"] = checkins
    out["Dep"] = c7.apply(lambda x: _primeiros_n(x, 3))
    out["Arr"] = c8.apply(lambda x: _primeiros_n(x, 3))

    out["Checkout"] = checkouts
    out["AcVer"] = c13
    out["DD"] = ""
    out["CAT"] = c6
    out["Crew"] = ""

    # Base de auditoria (antes de filtrar)
    auditoria = pd.DataFrame(index=df.index)
    auditoria["src_col_0"] = c0
    auditoria["src_col_0_ffill"] = c0_base
    auditoria["src_col_2"] = c2
    auditoria["src_col_3"] = c3
    auditoria["src_col_4"] = c4
    auditoria["src_col_6"] = c6
    auditoria["src_col_7"] = c7
    auditoria["src_col_8"] = c8
    auditoria["src_col_9"] = c9
    auditoria["src_col_13"] = c13
    auditoria["data_convertida_col0"] = data_conv
    auditoria["c3_hora"] = c3_hora
    auditoria["c3_dias"] = c3_dias
    auditoria["c7_hora"] = c7_hora
    auditoria["c7_dias"] = c7_dias
    auditoria["c8_hora"] = c8_hora
    auditoria["c8_dias"] = c8_dias
    auditoria["c9_hora"] = c9_hora
    auditoria["c9_dias"] = c9_dias
    auditoria["start_dias_usado"] = start_dias
    auditoria["end_dias_usado"] = end_dias
    auditoria["checkout_dias_usado"] = checkout_dias

    # Mantém linhas com qualquer conteúdo útil.
    # A data-base propagada preserva linhas de continuação do mesmo dia.
    mask_keep = (
        data_conv.astype(str).str.strip().ne("") |
        c2.astype(str).str.strip().ne("") |
        c3.astype(str).str.strip().ne("") |
        c4.astype(str).str.strip().ne("") |
        c6.astype(str).str.strip().ne("") |
        c7.astype(str).str.strip().ne("") |
        c8.astype(str).str.strip().ne("") |
        c9.astype(str).str.strip().ne("") |
        c13.astype(str).str.strip().ne("")
    )
    out = out[mask_keep].copy()
    auditoria = auditoria[mask_keep].copy()

    # Resultado final para conferência lado a lado
    for c in CSV_COLUMNS:
        auditoria[f"out_{c}"] = out[c]

    out = out[CSV_COLUMNS].reset_index(drop=True)
    auditoria = auditoria.reset_index(drop=True)

    if return_auditoria:
        return out, auditoria
    return out


def _eh_linha_numeracao_colunas(row_values) -> bool:
    """Detecta linha artificial como: 0;1;2;3;..."""
    vals = [str(v).strip() for v in row_values]
    vals = [v for v in vals if v != ""]
    if len(vals) < 3:
        return False

    if all(v.isdigit() for v in vals):
        nums = [int(v) for v in vals]
        # Aceita sequência crescente (0..N, 1..N, etc.)
        diffs = [b - a for a, b in zip(nums, nums[1:])]
        if nums == list(range(nums[0], nums[0] + len(nums))) and all(d == 1 for d in diffs):
            return True

    return False


def _eh_linha_cabecalho_rodape_latam(row_values) -> bool:
    txt = " ".join(str(v).strip() for v in row_values if str(v).strip())
    if not txt:
        return False
    up = txt.upper()

    return any([
        "ROSTER REPORT" in up,
        "PRINTED" in up,
        "PAGE" in up and "OF" in up,
        "UNNAMED:" in up,
        "MATRÍCULA" in up,
        "MATRICULA" in up,
        "CAUDURO" in up,
        "JAEGER" in up,
        bool(re.search(r"^\s*\d{1,2}-[A-Z]{3}-\d{4}\s*$", up)),
    ])


def _limpar_tabela_bruta_latam(df: pd.DataFrame) -> pd.DataFrame:
    """Limpa ruídos de extração e mantém a estrutura por posição de coluna."""
    if df is None or df.empty:
        return pd.DataFrame()

    x = df.copy()
    x = x.fillna("").astype(str)
    #x = x.applymap(lambda v: v.replace("\r", " ").replace("\n", " ").strip())

    x = x.map(lambda v: v.replace("\r", " ").replace("\n", " ").strip())

    # Preserva a posição original das colunas extraídas pelo Tabula.
    # Não removemos colunas vazias intermediárias porque isso desloca os índices.
    if x.shape[1] == 0:
        return pd.DataFrame()

    # Reindexa colunas para posição fixa 0..N
    x.columns = list(range(len(x.columns)))

    # Remove linhas totalmente vazias
    mask_not_empty = x.apply(lambda r: any(str(v).strip() for v in r), axis=1)
    x = x[mask_not_empty].copy()
    if x.empty:
        return x

    # Remove linha artificial de numeração de colunas e cabeçalho/rodapé de página
    mask_keep = []
    for _, row in x.iterrows():
        vals = row.tolist()
        if _eh_linha_numeracao_colunas(vals):
            mask_keep.append(False)
            continue
        if _eh_linha_cabecalho_rodape_latam(vals):
            mask_keep.append(False)
            continue
        mask_keep.append(True)

    x = x[pd.Series(mask_keep, index=x.index)].copy()
    x.reset_index(drop=True, inplace=True)
    return x


def _unificar_tabelas_alinhadas(tabelas: list[pd.DataFrame]) -> pd.DataFrame:
    """Une tabelas preservando a largura máxima para avaliação de alinhamento."""
    if not tabelas:
        return pd.DataFrame()

    max_cols = max((tab.shape[1] for tab in tabelas if tab is not None and not tab.empty), default=0)
    if max_cols == 0:
        return pd.DataFrame()

    blocos = []
    for tab in tabelas:
        if tab is None or tab.empty:
            continue
        t = tab.copy().fillna("").astype(str)
        t = t.reindex(columns=range(max_cols), fill_value="")
        blocos.append(t)

    if not blocos:
        return pd.DataFrame()

    df_unificado = pd.concat(blocos, ignore_index=True)
    df_unificado.columns = [f"col_{i}" for i in range(df_unificado.shape[1])]
    return df_unificado


def _ler_tabelas_pdf(pdf_path: str):
    """Lê PDF via tabula com tentativas em lattice e stream."""
    # Tentativa 1: lattice (melhor para tabela com borda)
    data = tabula.read_pdf(
        pdf_path,
        lattice=True,
        encoding="utf-8",
        pages="all",
        multiple_tables=True,
        pandas_options={"dtype": str, "header": None},
    )
    if data and len(data) > 0:
        return data

    # Tentativa 2: stream (melhor para layout sem borda)
    data = tabula.read_pdf(
        pdf_path,
        stream=True,
        guess=True,
        encoding="utf-8",
        pages="all",
        multiple_tables=True,
        pandas_options={"dtype": str, "header": None},
    )
    if data and len(data) > 0:
        return data

    return []


def _normalizar_datas_serie(series: pd.Series) -> pd.Series:
    def _fix(v):
        if pd.isna(v):
            return ""
        s = str(v).strip()
        if s in {"-", ""}:
            return s

        # Corrige casos invertidos como: '04:20 17/05/2026'
        m_inv = re.match(r"^(\d{1,2}:\d{2})\s+(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})$", s)
        if m_inv:
            s = f"{m_inv.group(2)} {m_inv.group(1)}"

        s_up = s.upper()
        repl = {
            "DEZ": "DEC", "FEV": "FEB", "ABR": "APR", "MAI": "MAY",
            "AGO": "AUG", "SET": "SEP", "OUT": "OCT",
        }
        for k, x in repl.items():
            s_up = s_up.replace(k, x)
        return s_up

    corrected = series.apply(_fix)

    fmts = [
        "%d/%m/%Y %H:%M", "%d-%m-%Y %H:%M", "%d/%m/%y %H:%M", "%d-%m-%y %H:%M",
        "%d%b%Y %H:%M", "%d%b%y %H:%M",
        "%Y-%m-%d %H:%M", "%Y/%m/%d %H:%M",
    ]

    converted = pd.Series(pd.NaT, index=corrected.index, dtype="datetime64[ns]")
    parse_mask = corrected.notna() & (corrected != "-") & (corrected != "")

    for fmt in fmts:
        mask = parse_mask & converted.isna()
        if mask.any():
            conv = pd.to_datetime(corrected[mask], format=fmt, errors="coerce")
            converted.loc[mask & conv.notna()] = conv[conv.notna()]

    mask = parse_mask & converted.isna()
    if mask.any():
        conv = pd.to_datetime(corrected[mask], errors="coerce", dayfirst=True)
        converted.loc[mask & conv.notna()] = conv[conv.notna()]

    out = corrected.astype("object")
    ok = converted.notna()
    if ok.any():
        out.loc[ok] = converted.loc[ok].dt.strftime("%d/%m/%Y %H:%M")

    return out


def _ajustar_datas_do_dr(df: pd.DataFrame) -> pd.DataFrame:
    """Corrige DO/DR quando Checkin/Start ficam após End/Checkout.

    Regra prática: para atividades `DO` e `DR`, se `Checkin` ou `Start`
    ficarem com data/hora maior que a referência final da linha (`End`/`Checkout`),
    recua 1 dia até não ultrapassar a referência.
    """
    if df.empty or "Activity" not in df.columns:
        return df

    out = df.copy()
    atividade = out["Activity"].fillna("").astype(str).str.strip().str.upper()
    mask_do_dr = atividade.isin(["DO", "DR"])
    if not mask_do_dr.any():
        return out

    def _parse_col(col_name: str) -> pd.Series:
        if col_name not in out.columns:
            return pd.Series(pd.NaT, index=out.index, dtype="datetime64[ns]")
        return pd.to_datetime(out[col_name], errors="coerce", dayfirst=True)

    dt_checkin = _parse_col("Checkin")
    dt_start = _parse_col("Start")
    dt_end = _parse_col("End")
    dt_checkout = _parse_col("Checkout")

    ref_final = dt_end.copy()
    usar_checkout = ref_final.isna() | (dt_checkout.notna() & (dt_checkout < ref_final))
    ref_final.loc[usar_checkout] = dt_checkout.loc[usar_checkout]

    for col_name, dt_col in [("Checkin", dt_checkin), ("Start", dt_start)]:
        valores = dt_col.copy()
        mask = mask_do_dr & valores.notna() & ref_final.notna() & (valores > ref_final)
        if not mask.any():
            continue

        while mask.any():
            valores.loc[mask] = valores.loc[mask] - pd.Timedelta(days=1)
            mask = mask_do_dr & valores.notna() & ref_final.notna() & (valores > ref_final)

        out.loc[valores.notna(), col_name] = valores.loc[valores.notna()].dt.strftime("%d/%m/%Y %H:%M")

    return out


def _corrigir_continuidade_jornadas(df: pd.DataFrame) -> pd.DataFrame:
    """Garante coerência temporal em linhas sem Checkin (continuação de jornada).

    Regras:
    - Se `Checkin` estiver vazio, então `Start` deve ser > `End` anterior.
    - `End` não pode ser menor que `Start`.
    - `Checkout` não pode ser menor que `End`.
    """
    if df.empty:
        return df

    out = df.copy()

    def _to_dt(v: str):
        return pd.to_datetime(v, errors="coerce", dayfirst=True)

    prev_end = pd.NaT
    for idx in out.index:
        act = str(out.at[idx, "Activity"]).strip().upper() if "Activity" in out.columns else ""
        ch = str(out.at[idx, "Checkin"]).strip() if "Checkin" in out.columns else ""

        ch_dt = _to_dt(ch) if ch else pd.NaT
        st = _to_dt(out.at[idx, "Start"]) if "Start" in out.columns else pd.NaT
        en = _to_dt(out.at[idx, "End"]) if "End" in out.columns else pd.NaT
        co = _to_dt(out.at[idx, "Checkout"]) if "Checkout" in out.columns else pd.NaT

        # Se há checkin válido (ex.: DR), ele não pode ficar anterior ao End da linha anterior.
        # Quando isso ocorrer, avança a linha inteira em dias para manter coerência.
        if pd.notna(ch_dt) and pd.notna(prev_end):
            delta_days = 0
            while ch_dt < prev_end:
                ch_dt += pd.Timedelta(days=1)
                delta_days += 1
            if delta_days > 0:
                if pd.notna(st):
                    st += pd.Timedelta(days=delta_days)
                if pd.notna(en):
                    en += pd.Timedelta(days=delta_days)
                if pd.notna(co):
                    co += pd.Timedelta(days=delta_days)

        # Dentro da linha, Start deve ser >= Checkin quando ambos existem.
        if pd.notna(ch_dt) and pd.notna(st):
            delta_days = 0
            while st < ch_dt:
                st += pd.Timedelta(days=1)
                delta_days += 1
            if delta_days > 0:
                if pd.notna(en):
                    en += pd.Timedelta(days=delta_days)
                if pd.notna(co):
                    co += pd.Timedelta(days=delta_days)

        # Continuação de jornada (sem checkin)
        if ch == "" and pd.notna(prev_end) and pd.notna(st):
            delta_days = 0
            while st <= prev_end:
                st += pd.Timedelta(days=1)
                delta_days += 1
            if delta_days > 0:
                if pd.notna(en):
                    en += pd.Timedelta(days=delta_days)
                if pd.notna(co):
                    co += pd.Timedelta(days=delta_days)

        # Ordem temporal dentro da mesma linha
        if pd.notna(st) and pd.notna(en):
            while en < st:
                en += pd.Timedelta(days=1)

        if pd.notna(en) and pd.notna(co):
            while co < en:
                co += pd.Timedelta(days=1)

        if pd.notna(st) and "Start" in out.columns:
            out.at[idx, "Start"] = st.strftime("%d/%m/%Y %H:%M")
        if pd.notna(ch_dt) and "Checkin" in out.columns:
            out.at[idx, "Checkin"] = ch_dt.strftime("%d/%m/%Y %H:%M")
        if pd.notna(en) and "End" in out.columns:
            out.at[idx, "End"] = en.strftime("%d/%m/%Y %H:%M")
        if pd.notna(co) and "Checkout" in out.columns:
            out.at[idx, "Checkout"] = co.strftime("%d/%m/%Y %H:%M")

        if pd.notna(en):
            prev_end = en

    return out


def _limpar_e_ajustar(df: pd.DataFrame) -> pd.DataFrame:
    df = df.copy()
    df.replace("\r", " ", regex=True, inplace=True)
    df.replace("- nan", "-", regex=True, inplace=True)
    df.dropna(how="all", inplace=True)
    df.reset_index(drop=True, inplace=True)

    # Remove linhas de total/rodapé
    if "Activity" in df.columns:
        a = df["Activity"].astype(str).str.upper().str.strip()
        df = df[~a.isin(["TOTAL:", "TOTAL", "OCULTO", "NAN"])].copy()

    # Ajustes padrão de checkin/checkout
    if {"Activity", "Checkin", "Start"}.issubset(df.columns):
        activity_upper = df["Activity"].astype(str).str.upper()
        mask = (df["Checkin"].astype(str).str.strip() == "-") & (~activity_upper.str.startswith(("AD", "LA"), na=False))
        df.loc[mask, "Checkin"] = df.loc[mask, "Start"]

    if {"Checkout", "End"}.issubset(df.columns):
        m2 = df["Checkout"].astype(str).str.strip().isin(["", "-"])
        df.loc[m2, "Checkout"] = df.loc[m2, "End"]

    # Normaliza todas as colunas de data/hora e corrige inversões data<->hora.
    for col in ["Checkin", "Start", "End", "Checkout"]:
        if col in df.columns:
            df[col] = _normalizar_datas_serie(df[col])

    df = _ajustar_datas_do_dr(df)
    df = _corrigir_continuidade_jornadas(df)

    # Garante colunas finais
    for c in CSV_COLUMNS:
        if c not in df.columns:
            df[c] = ""

    return df[CSV_COLUMNS].reset_index(drop=True)


# ---------------------------
# Nova função: remover colunas totalmente vazias
# ---------------------------
def drop_colunas_totalmente_vazias(df: pd.DataFrame, treat_as_empty=None) -> pd.DataFrame:
    """
    Remove colunas cujo conteúdo, em todas as linhas, é considerado "vazio".
    Parâmetros:
      - df: DataFrame de entrada.
      - treat_as_empty: lista de strings adicionais a tratar como vazias (ex: ["-", "nan"]).
    Retorna:
      - DataFrame com colunas totalmente vazias removidas.
    """
    if df is None or df.empty:
        return df.copy()

    treat_as_empty = set(treat_as_empty or [])
    # Normaliza cada célula para string e verifica se é vazia após strip
    def _is_empty_cell(x):
        if pd.isna(x):
            return True
        s = str(x).strip()
        if s == "":
            return True
        if s in treat_as_empty:
            return True
        return False

    # Cria máscara booleana por coluna: True se todas as células são "vazias"
    cols_para_remover = []
    for col in df.columns:
        col_vals = df[col].apply(_is_empty_cell)
        if col_vals.all():
            cols_para_remover.append(col)

    if cols_para_remover:
        # Log simples para auditoria
        print(f"Removendo colunas totalmente vazias: {cols_para_remover}")
        return df.drop(columns=cols_para_remover)
    return df.copy()


# ---------------------------
# Nova função: detectar '<==' e ajustar início do mês (data_ref - 1 dia)
# ---------------------------
def ajustar_inicio_mes_por_marcador(df: pd.DataFrame) -> pd.DataFrame:
    """
    Se a coluna 0 linha 0 contém '<==', aplica a lógica:
      1) encontra a primeira data explícita na coluna 0 (ex: 02-Jun-2026)
      2) armazena essa data como referência
      3) retrocede até encontrar a linha com hora válida na coluna 3
      4) calcula data_inicial = referencia - 1 dia (sem procurar dia 1)
      5) atribui data_inicial (formatada dd/mm/YYYY) às linhas do bloco (desde a linha encontrada até a linha anterior à data explícita)
    Retorna o DataFrame com a coluna 0 ajustada (mantendo demais colunas).
    """
    if df is None or df.empty:
        return df

    x = df.copy().reset_index(drop=True)

    # garante existência da coluna 0
    if 0 not in x.columns:
        return x

    primeira_celula = str(x.at[0, 0]).strip() if 0 in x.columns else ""
    if primeira_celula != "<==":
        return x

    # 1) encontrar primeira linha com data válida na coluna 0
    idx_data = None
    data_ref_dt = None
    for i in range(len(x)):
        raw = str(x.at[i, 0]).strip()
        norm = _normalizar_data_base(raw)
        if norm:
            try:
                data_ref_dt = datetime.strptime(norm, "%d/%m/%Y")
                idx_data = i
                break
            except Exception:
                continue

    if idx_data is None or data_ref_dt is None:
        return x

    # 2) retroceder até encontrar linha com hora válida na coluna 3
    idx_hora = None
    for j in range(idx_data - 1, -1, -1):
        raw3 = str(x.at[j, 3]) if 3 in x.columns else ""
        hh, dias, bruto = _parse_hora_e_deslocamento(raw3)
        if hh:
            idx_hora = j
            break

    if idx_hora is None:
        return x

    # 3) calcular data inicial: referencia - 1 dia (sem procurar dia 1)
    candidate = data_ref_dt - timedelta(days=1)
    data_inicial_str = candidate.strftime("%d/%m/%Y")

    # 4) atribuir data_inicial às linhas do bloco: desde idx_hora até idx_data-1 (inclusive)
    start_idx = idx_hora
    end_idx = idx_data  # exclusive
    for k in range(start_idx, end_idx):
        x.at[k, 0] = data_inicial_str

    return x
